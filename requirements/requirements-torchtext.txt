# torchtext 전용 환경 (.venv-torchtext) — CPU 전용
#
# 왜 별도 환경인가:
#   torchtext는 개발이 종료되어 0.18.0(torch 2.3.x)이 마지막 버전입니다.
#   컴파일된 확장(libtorchtext)이 torch 2.3의 C++ ABI에 링크되어 있어
#   메인 환경(.venv, torch 2.7.1+cu128)에서는 import 시 WinError 127로 실패합니다.
#   또한 RTX 5060(sm_120)은 cu121 빌드가 지원하지 않으므로 이 환경은 CPU 전용입니다.
#   → GPU 학습은 메인 .venv, torchtext 전처리/데이터셋은 이 환경에서 사용하세요.
#
# 설치:
#   uv venv --python 3.11 .venv-torchtext
#   uv pip install --python .venv-torchtext\Scripts\python.exe --index-url https://download.pytorch.org/whl/cpu --extra-index-url https://pypi.org/simple -r requirements-torchtext.txt
#   .venv-torchtext\Scripts\python.exe -m ipykernel install --user --name torchtext --display-name "Python 3.11 (torchtext, CPU)"
#
# 노트북에서: 커널 선택 → "Python 3.11 (torchtext, CPU)"

# --- 핵심 (버전 고정 필수: 셋 중 하나라도 어긋나면 import 실패) ---
torch==2.3.0
torchtext==0.18.0
torchdata==0.8.0   # 0.7.1은 torch 2.3에서 DILL_AVAILABLE ImportError 발생
numpy<2            # torch 2.3은 numpy 2.x와 호환되지 않음
portalocker        # torchtext.datasets 다운로드 캐시에 필요

# --- spaCy 토크나이저 (7장 Multi30k 독→영 번역에서 사용) ---
spacy==3.7.5
click==8.5.0   # spacy CLI가 요구하지만 의존성에 빠져 있어 명시 필요
de_core_news_sm @ https://github.com/explosion/spacy-models/releases/download/de_core_news_sm-3.7.0/de_core_news_sm-3.7.0-py3-none-any.whl
en_core_web_sm @ https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.7.1/en_core_web_sm-3.7.1-py3-none-any.whl

ipykernel
